## 1200. Minimum Absolute Difference

## 나의 풀이
```java
import java.util.Arrays;

class Solution {
    public List<List<Integer>> minimumAbsDifference(int[] arr) {
        
        List<Integer> numArr = Arrays.stream(arr)
                            .boxed()
                            .sorted() 
                            .collect(Collectors.toList());
        int min = Integer.MAX_VALUE;
        List<List<Integer>> answer = new ArrayList<>();


        for(int i=0; i<numArr.size(); i++) {
            if(i > 0) {
                int bef = numArr.get(i-1);
                int aft = numArr.get(i);
                int diff = Math.abs(aft - bef);

                if(diff < min) {
                    min = diff;
                }

                List<Integer> data = new ArrayList<>();
                data.add(numArr.get(i-1));
                data.add(numArr.get(i));
                answer.add(data);
            }
        }
        
        // 🚫🚫🚫🚫min이 final이어야 함. final이면 min에 값 할당을 못함. javascript의 const같은
        // answer.removeIf(item -> min != Math.abs(item.get(1) - item.get(0)));

        // 🚫🚫🚫🚫index가 unshirft됨 주의
        // for(int i=0; i<answer.size(); i++) {
        //     int diff = Math.abs(answer.get(i).get(1) - answer.get(i).get(0));
            
        //     if(diff != min) {
        //         answer.remove(i);
        //     }
        // }

        Iterator<List<Integer>> iterator = answer.iterator();

        while (iterator.hasNext()) {
            List<Integer> number = iterator.next();
            int diff = Math.abs(number.get(1) - number.get(0));

            if (diff != min) {
                iterator.remove();
            }
        }

        return answer;
    }
}
```

## Solution
```java
class Solution {
  public List<List<Integer>> minimumAbsDifference(int[] arr) {
    List<List<Integer>> ans = new ArrayList<>();
    int mn = Integer.MAX_VALUE;

    Arrays.sort(arr);

    for (int i = 1; i < arr.length; ++i) {
      final int diff = arr[i] - arr[i - 1];
      if (diff < mn) {
        mn = diff;
        ans.clear();
      }
      if (diff == mn)
        ans.add(Arrays.asList(arr[i - 1], arr[i]));
    }

    return ans;
  }
}
```
